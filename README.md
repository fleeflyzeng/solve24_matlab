>> solve24([2 2 3 4])

  (((2*2)+4)*3) = 24
  (((2+2)+4)*3) = 24
  (((2+4)+2)*3) = 24
  (((4+2)+2)*3) = 24
  ((2+(2+4))*3) = 24
  ((2+(4+2))*3) = 24
  ((4+(2*2))*3) = 24
  ((4+(2+2))*3) = 24
  (3*((2*2)+4)) = 24
  (3*((2+2)+4)) = 24
  (3*((2+4)+2)) = 24
  (3*((4+2)+2)) = 24
  (3*(2+(2+4))) = 24
  (3*(2+(4+2))) = 24
  (3*(4+(2*2))) = 24
  (3*(4+(2+2))) = 24
